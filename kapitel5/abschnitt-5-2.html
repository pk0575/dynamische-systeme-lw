5.2Wandernde Wellen

Auf einer Faser läuft die Erregung als Puls mit fester Form und Geschwindigkeit. Für die Front lässt sich die Geschwindigkeit c = √(D/(2τin)) (1 − 3vs*) exakt angeben; sie wächst mit der Wurzel aus der Kopplung D. Die Programme verwenden D = 0.1 mm2/ms, Δx = 0.1 mm und Δt = 0.01 ms.

Zu jeder Abbildung des Abschnitts finden Sie das Programm, das sie erzeugt. Ändern Sie Parameter und Startwerte und beobachten Sie, was passiert. Abbildungs- und Gleichungsnummern beziehen sich auf das Buch.

Abb. 5.2

Ein Puls auf der Faser

∂v/∂t = D ∂2v/∂x2 + h v2 (1 − v)/τin − v/τout + J(x, t)

Reiz am linken Ende einer 150 mm langen Faser. Aus den Zeitpunkten, zu denen v bei 25 und 75 mm den Wert 0.5 erreicht, ergibt sich c ≈ 0.34 mm/ms; mit APD ≈ 227 ms ist die Wellenlänge etwa 77 mm.

# Abb. 5.2: Ein Puls auf der Faser (Buch, Abschnitt 5.2, Gl. 5.2)
# Kabelgleichung mit dem Mitchell-Schaeffer-Modell, Faser 150 mm, Reiz am linken Ende; Linienmethode mit dem Euler-Verfahren
# (c) 2026 Philipp Kügler · Programm zum Buch „Dynamische Systeme in den Lebenswissenschaften“ · MIT-Lizenz (siehe LICENSE)
using Printf, Plots
tin = 0.3; tout = 6.0; topen = 100.0; tclose = 115.0; vgate = 0.13   # Parameter in ms
D = 0.1; dx = 0.1; dt = 0.01                                         # D in mm^2/ms, Gitterabstand in mm, Zeitschritt in ms
L = 150.0; T = 500.0
N = round(Int, L / dx) + 1; x = dx .* (0:N-1)
v = zeros(N); h = ones(N)
reiz = x .<= 1.0                                                     # Reiz J = 0.2 auf dem ersten Millimeter für 2 ms
zeiten = [100.0, 250.0, 400.0]; schnapp = Dict{Int, Vector{Float64}}()
K = round(Int, T / dt); s25 = zeros(K); s75 = zeros(K); t_akt = fill(NaN, N)
lap = zeros(N)
for n in 0:K-1
    t = n * dt
    for z in zeiten
        abs(t - z) < 1e-6 && (schnapp[round(Int, z)] = copy(v))
    end
    lap[2:end-1] .= v[3:end] .- 2 .* v[2:end-1] .+ v[1:end-2]
    lap[1] = v[2] - v[1]; lap[end] = v[end-1] - v[end]               # Rand ohne Fluss
    dv = h .* v.^2 .* (1 .- v) ./ tin .- v ./ tout .+ D .* lap ./ dx^2
    t < 2.0 && (dv[reiz] .+= 0.2)
    dh = ifelse.(v .< vgate, (1 .- h) ./ topen, -h ./ tclose)
    global v = v .+ dt .* dv; global h = h .+ dt .* dh
    neu = isnan.(t_akt) .& (v .> 0.5); t_akt[neu] .= t + dt          # Zeitpunkt, an dem v erstmals 0.5 erreicht
    s25[n + 1] = v[251]; s75[n + 1] = v[751]
end
c = (x[751] - x[251]) / (t_akt[751] - t_akt[251])
ts = dt .* collect(1:K); a = ts[s75 .> 0.1]; apd = a[end] - a[1]
@printf("v = 0.5 bei x = 25 mm nach %.1f ms, bei x = 75 mm nach %.1f ms\n", t_akt[251], t_akt[751])
@printf("Leitungsgeschwindigkeit c = %.4f mm/ms,  APD = %.1f ms,  Wellenlänge c APD = %.1f mm\n", c, apd, c * apd)
farben = [RGB(0.89, 0.44, 0.49), RGB(0.64, 0.08, 0.18), RGB(0.43, 0.04, 0.12)]
p1 = plot(xlabel = "x (mm)", ylabel = "v(x, t)")
for (j, z) in enumerate((100, 250, 400))
    plot!(p1, x, schnapp[z], color = farben[j], lw = 2, label = "t = $z ms")
end
p2 = plot(ts, s25, color = farben[3], lw = 2, label = "x = 25 mm", xlabel = "t (ms)", ylabel = "v(x, t)")
plot!(p2, ts, s75, color = farben[1], lw = 2, label = "x = 75 mm")
display(plot(p1, p2, layout = (1, 2), size = (900, 420)))

Abb. 5.3

Leitungsgeschwindigkeit und Kopplung

c = √(D/(2τin)) (1 − 3vs*)

Simulationen für fünf Werte von D und die Formel. Die Simulation liegt etwa 1 % darunter, weil sich die Kanäle schon während des Anstiegs zu schließen beginnen.

# Abb. 5.3: Leitungsgeschwindigkeit und Kopplung (Buch, Abschnitt 5.2, Gl. 5.5)
# Simulation der Kabelgleichung für fünf Werte von D und die Formel c = sqrt(D/(2 tin)) (1 - 3 vs)
# (c) 2026 Philipp Kügler · Programm zum Buch „Dynamische Systeme in den Lebenswissenschaften“ · MIT-Lizenz (siehe LICENSE)
using Printf, Plots
tin = 0.3; tout = 6.0; topen = 100.0; tclose = 115.0; vgate = 0.13   # Parameter in ms
hc = 4 * tin / tout
vs = (1 - sqrt(1 - hc)) / 2                                          # Schwelle für h = 1
formel(D) = sqrt(D / (2 * tin)) * (1 - 3 * vs)

function geschwindigkeit(D; dx = 0.1, dt = 0.01, L = 60.0)
    # Faser der Länge L, Reiz am linken Ende; c aus den Zeitpunkten, an denen v bei 20 und 50 mm den Wert 0.5 erreicht
    N = round(Int, L / dx) + 1; v = zeros(N); h = ones(N); reiz = dx .* (0:N-1) .<= 1.0
    i1 = round(Int, 20 / dx) + 1; i2 = round(Int, 50 / dx) + 1; t1 = NaN; t2 = NaN; n = 0; lap = zeros(N)
    while isnan(t2)
        lap[2:end-1] .= v[3:end] .- 2 .* v[2:end-1] .+ v[1:end-2]; lap[1] = v[2] - v[1]; lap[end] = v[end-1] - v[end]
        dv = h .* v.^2 .* (1 .- v) ./ tin .- v ./ tout .+ D .* lap ./ dx^2
        n * dt < 2.0 && (dv[reiz] .+= 0.2)
        dh = ifelse.(v .< vgate, (1 .- h) ./ topen, -h ./ tclose)
        v = v .+ dt .* dv; h = h .+ dt .* dh; n += 1
        isnan(t1) && v[i1] > 0.5 && (t1 = n * dt)
        isnan(t2) && v[i2] > 0.5 && (t2 = n * dt)
    end
    return 30.0 / (t2 - t1)
end

cs = [geschwindigkeit(D) for D in Ds]
for (D, c) in zip(Ds, cs)
    @printf("D = %5.3f:  Simulation c = %.4f,  Formel c = %.4f mm/ms\n", D, c, formel(D))
end
Dd = range(0, 0.45, length = 300)
plot(Dd, formel.(Dd), color = RGB(0, 0.447, 0.741), lw = 2.5, label = "Formel", xlabel = "D (mm²/ms)", ylabel = "c (mm/ms)", legend = :bottomright)
display(scatter!(Ds, cs, color = RGB(0.64, 0.08, 0.18), ms = 6, label = "Simulation"))

def geschwindigkeit(D, dx=0.1, dt=0.01, L=60.0):
    """Faser der Länge L, Reiz am linken Ende; c aus den Zeitpunkten, an denen v bei 20 und 50 mm den Wert 0.5 erreicht."""
    N = int(round(L / dx)) + 1; v = np.zeros(N); h = np.ones(N); reiz = dx * np.arange(N) <= 1.0
    i1, i2 = int(round(20 / dx)), int(round(50 / dx)); t1 = t2 = None; n = 0
    while t2 is None:
        lap = np.empty(N); lap[1:-1] = v[2:] - 2 * v[1:-1] + v[:-2]; lap[0] = v[1] - v[0]; lap[-1] = v[-2] - v[-1]

        if t1 is None and v[i1] > 0.5: t1 = n * dt
        if t2 is None and v[i2] > 0.5: t2 = n * dt
    return 30.0 / (t2 - t1)

geschwindigkeit <- function(D, dx = 0.1, dt = 0.01, L = 60) {
  # Faser der Länge L, Reiz am linken Ende; c aus den Zeitpunkten, an denen v bei 20 und 50 mm den Wert 0.5 erreicht
  N <- round(L / dx) + 1; v <- numeric(N); h <- rep(1, N); reiz <- dx * (0:(N - 1)) <= 1; innen <- 2:(N - 1)
  i1 <- round(20 / dx) + 1; i2 <- round(50 / dx) + 1; t1 <- NA; t2 <- NA; n <- 0
  while (is.na(t2)) {
    lap <- c(v[2] - v[1], v[innen + 1] - 2 * v[innen] + v[innen - 1], v[N - 1] - v[N])
    dv <- h * v^2 * (1 - v) / tin - v / tout + D * lap / dx^2
    if (n * dt < 2) dv[reiz] <- dv[reiz] + 0.2
    dh <- ifelse(v < vgate, (1 - h) / topen, -h / tclose)
    v <- v + dt * dv; h <- h + dt * dh; n <- n + 1
    if (is.na(t1) && v[i1] > 0.5) t1 <- n * dt
    if (is.na(t2) && v[i2] > 0.5) t2 <- n * dt
  }
  30 / (t2 - t1)
}

Ds <- c(0.025, 0.05, 0.1, 0.2, 0.4)
cs <- sapply(Ds, geschwindigkeit)
for (k in seq_along(Ds)) cat(sprintf("D = %5.3f:  Simulation c = %.4f,  Formel c = %.4f mm/ms\n", Ds[k], cs[k], formel(Ds[k])))
Dd <- seq(0, 0.45, length.out = 300)
plot(Dd, formel(Dd), type = "l", col = rgb(0, 0.447, 0.741), lwd = 2.5, ylim = c(0, 0.8), xlab = "D (mm²/ms)", ylab = "c (mm/ms)")
points(Ds, cs, pch = 19, col = rgb(0.64, 0.08, 0.18), cex = 1.5)
legend("bottomright", legend = c("Formel", "Simulation"), col = c(rgb(0, 0.447, 0.741), rgb(0.64, 0.08, 0.18)), lwd = c(2.5, NA), pch = c(NA, 19))
